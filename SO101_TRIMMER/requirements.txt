# wheel for Python 3.14 yet, which is the single best reason to run this on
# 3.11 or 3.12 rather than on Thonny's bundled interpreter.
onnxruntime
pandas                 # the dashboard's tables

# --- only for the CAD ------------------------------------------------------

trimesh
manifold3d
